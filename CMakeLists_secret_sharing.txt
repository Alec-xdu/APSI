cmake_minimum_required(VERSION 3.16)
project(SecretSharingPSI VERSION 1.0.0 LANGUAGES CXX)

# 设置C++标准
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 查找OpenSSL
find_package(OpenSSL REQUIRED)

# 查找SEAL库（如果使用）
# find_package(SEAL REQUIRED)

# 设置编译选项
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra -O2")

# 包含目录
include_directories(
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${OPENSSL_INCLUDE_DIR}
)

# 源文件
set(COMMON_SOURCES
    common/apsi/secret_sharing.cpp
    common/apsi/tee_manager.cpp
)

set(RECEIVER_SOURCES
    receiver/apsi/secret_sharing_powers.cpp
)

set(EXAMPLE_SOURCES
    examples/secret_sharing_psi_example.cpp
)

# 创建库
add_library(apsi_secret_sharing STATIC ${COMMON_SOURCES})
target_link_libraries(apsi_secret_sharing ${OPENSSL_LIBRARIES})

add_library(apsi_receiver_secret_sharing STATIC ${RECEIVER_SOURCES})
target_link_libraries(apsi_receiver_secret_sharing apsi_secret_sharing)

# 创建示例程序
add_executable(secret_sharing_psi_example ${EXAMPLE_SOURCES})
target_link_libraries(secret_sharing_psi_example 
    apsi_secret_sharing 
    apsi_receiver_secret_sharing
    ${OPENSSL_LIBRARIES}
)

# 安装规则
install(TARGETS apsi_secret_sharing apsi_receiver_secret_sharing
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
)

install(TARGETS secret_sharing_psi_example
    RUNTIME DESTINATION bin
)

# 安装头文件
install(FILES 
    common/apsi/secret_sharing.h
    common/apsi/tee_manager.h
    receiver/apsi/secret_sharing_powers.h
    DESTINATION include/apsi
)

# 测试
enable_testing()
add_executable(test_secret_sharing tests/test_secret_sharing.cpp)
target_link_libraries(test_secret_sharing 
    apsi_secret_sharing 
    apsi_receiver_secret_sharing
    ${OPENSSL_LIBRARIES}
)
add_test(NAME SecretSharingTest COMMAND test_secret_sharing)
